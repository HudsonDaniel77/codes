#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

sem_t mutex, wrt;
int readcount = 0;
int data = 0;

void wait_sem(sem_t *s) {
    sem_wait(s);
}

void signal_sem(sem_t *s) {
    sem_post(s);
}

void *reader(void *arg) {
    int id = *(int *)arg;

    wait_sem(&mutex);
    readcount++;
    if (readcount == 1)
        wait_sem(&wrt);
    signal_sem(&mutex);

    printf("Reader %d reads data = %d\n", id, data);
    sleep(1);

    wait_sem(&mutex);
    readcount--;
    if (readcount == 0)
        signal_sem(&wrt);
    signal_sem(&mutex);

    return NULL;
}

void *writer(void *arg) {
    int id = *(int *)arg;

    wait_sem(&wrt);
    data++;
    printf("Writer %d writes data = %d\n", id, data);
    sleep(1);
    signal_sem(&wrt);

    return NULL;
}

int main() {
    pthread_t r[2], w[2];
    int id[2] = {1, 2};

    sem_init(&mutex, 0, 1);
    sem_init(&wrt, 0, 1);

    for (int i = 0; i < 2; i++)
        pthread_create(&r[i], NULL, reader, &id[i]);

    for (int i = 0; i < 2; i++)
        pthread_create(&w[i], NULL, writer, &id[i]);

    for (int i = 0; i < 2; i++) {
        pthread_join(r[i], NULL);
        pthread_join(w[i], NULL);
    }

    sem_destroy(&mutex);
    sem_destroy(&wrt);
    return 0;
}
