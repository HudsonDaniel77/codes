
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <sys/wait.h>
#include <unistd.h>

struct message
{
    long type;
    char text[100];
};

int main()
{
    int msgid;
    struct message msg;

    // Create message queue
    msgid = msgget(IPC_PRIVATE, IPC_CREAT | 0666);

    if (msgid == -1)
    {
        perror("msgget failed");
        return 1;
    }

    if (fork() == 0)
    {
        // RECEIVER (Child)
        printf("\n--- RECEIVER ---\n");

        msgrcv(msgid, &msg, sizeof(msg.text), 1, 0);

        printf("Received message: %s\n", msg.text);

        msgctl(msgid, IPC_RMID, NULL);
        exit(0);
    }
    else
    {
        // SENDER (Parent)
        printf("\n--- SENDER ---\n");

        msg.type = 1;

        printf("Enter message: ");
        scanf(" %99[^\n]", msg.text);

        msgsnd(msgid, &msg, sizeof(msg.text), 0);

        printf("Message sent successfully.\n");

        wait(NULL);
    }

    return 0;
}
