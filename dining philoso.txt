
#include <stdio.h>
#include <pthread.h>
#include <semaphore.h>
#include <unistd.h>

sem_t fork[5];

void *philosopher(void *arg) {
    int id = *(int *)arg;

    printf("Philosopher %d is thinking\n", id);

    sem_wait(&fork[id]);
    sem_wait(&fork[(id + 1) % 5]);

    printf("Philosopher %d is eating\n", id);
    sleep(1);

    sem_post(&fork[id]);
    sem_post(&fork[(id + 1) % 5]);

    printf("Philosopher %d finished eating\n", id);

    return NULL;
}

int main() {
    pthread_t p[5];
    int id[5];

    for (int i = 0; i < 5; i++) {
        sem_init(&fork[i], 0, 1);
        id[i] = i;
    }

    for (int i = 0; i < 5; i++)
        pthread_create(&p[i], NULL, philosopher, &id[i]);

    for (int i = 0; i < 5; i++)
        pthread_join(p[i], NULL);

    for (int i = 0; i < 5; i++)
        sem_destroy(&fork[i]);

    return 0;
}
