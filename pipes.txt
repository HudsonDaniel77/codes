
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
#include <string.h>

int main() {
    int p1[2], p2[2];
    char msg[] = "Hello Child";
    char reply[] = "Hello Parent";
    char buffer[100];

    pipe(p1);
    pipe(p2);

    if (fork() == 0) {
        close(p1[1]);
        close(p2[0]);

        read(p1[0], buffer, sizeof(buffer));
        printf("Child received: %s\n", buffer);

        write(p2[1], reply, strlen(reply) + 1);

        close(p1[0]);
        close(p2[1]);
    } else {
        close(p1[0]);
        close(p2[1]);

        write(p1[1], msg, strlen(msg) + 1);

        read(p2[0], buffer, sizeof(buffer));
        printf("Parent received: %s\n", buffer);

        close(p1[1]);
        close(p2[0]);

        wait(NULL);
    }

    return 0;
}
